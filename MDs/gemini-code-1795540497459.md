# MD 4 — AURION ADAPTIVE ROUTER & AUTONOMOUS DESKTOP RPA

## 1. Arquitetura do Roteamento Adaptativo & Controle Autônomo
Este documento especifica:
1. **Aurion Adaptive Router (`apps/vps-harness`):** Orquestrador multi-API com classificação semântica de tarefa, *cooldown* automático em Redis para erros `429`/`503`, *auto-scoring* de confiabilidade em SQLite e **trava rígida para a API do Claude** (acionada apenas como último recurso após 2 falhas consecutivas ou comando explícito, respeitando teto diário).
2. **Closed-Loop GUI Agent (`Observe → Think → Act → Verify`):** Worker na VPS que enxerga a tela do PC em coordenadas normalizadas (`0–1000`), decide cliques/digitação e valida o resultado visualmente.
3. **Módulo Nativo Rust RPA + Tecla de Pânico (`apps/desktop-eye`):** Execução local de mouse/teclado com curva suave e *kill-switch* global (`ESC` 2x em `< 600ms` ou `Ctrl+Alt+K`) que interrompe qualquer movimento em `0ms`.

import asyncio
import json
import time
from dataclasses import dataclass, field
from enum import Enum
from typing import Any, Dict, List, Optional

import httpx
import redis.asyncio as redis


class TaskType(str, Enum):
    FAST_VOICE = "FAST_VOICE"        # Respostas curtas, voz em tempo real, triagem
    VISION_GUI = "VISION_GUI"        # Leitura de screenshot da tela e coordenadas RPA
    CODING = "CODING"                # Geração de código, scripts, SQL, refatoração
    GENERAL = "GENERAL"              # Planejamento, resumos, automações gerais
    CRITICAL_RESCUE = "CRITICAL_RESCUE"  # Escalonamento manual ou pós-falha dupla


@dataclass
class ProviderNode:

    name: str
    base_url: str
    model: str
    api_key_vault_ref: str
    supported_tasks: List[TaskType]
    priority: int  # Menor número = maior prioridade
    is_last_resort: bool = False
    reliability_score: float = 100.0
    avg_latency_ms: float = 500.0


# Matriz padrão de provedores (chaves carregadas do Cofre AES-256 da VPS)

class AurionAdaptiveRouter:
    def __init__(self, redis_client: redis.Redis, vault_getter, claude_daily_cap: int = 15):
        self.redis = redis_client
        self.get_secret = vault_getter
        self.providers = DEFAULT_PROVIDERS
        self.claude_daily_cap = claude_daily_cap

    async def _is_in_cooldown(self, provider_id: str) -> bool:
        return await self.redis.exists(f"aurion:cooldown:{provider_id}") > 0

    async def _set_cooldown(self, provider_id: str, seconds: int = 60):
        await self.redis.setex(f"aurion:cooldown:{provider_id}", seconds, "1")

    async def _can_use_claude(self, failure_count: int, force_claude: bool) -> bool:
        if not force_claude and failure_count < 2:
            return False
        today_key = f"aurion:claude_calls:{time.strftime('%Y-%m-%d')}"
        calls = int(await self.redis.get(today_key) or 0)
        return calls < self.claude_daily_cap

    async def _increment_claude_usage(self):
        today_key = f"aurion:claude_calls:{time.strftime('%Y-%m-%d')}"
        pipe = self.redis.pipeline()
        pipe.incr(today_key)
        pipe.expire(today_key, 86400)
        await pipe.execute()

    def _update_score(self, provider: ProviderNode, success: bool, latency_ms: float):
        if success:
            provider.avg_latency_ms = (provider.avg_latency_ms * 0.8) + (latency_ms * 0.2)
            provider.reliability_score = min(100.0, provider.reliability_score + 1.5)
        else:
            provider.reliability_score = max(10.0, provider.reliability_score - 12.0)

    async def route_completion(
        self,
        messages: List[Dict[str, Any]],
        task_type: TaskType = TaskType.GENERAL,
        json_mode: bool = False,
        force_claude: bool = False,
        temperature: float = 0.2,
    ) -> Dict[str, Any]:
        # Filtra provedores compatíveis com a tarefa e ordena por prioridade + reliability_score
        candidates = [
            p for p in self.providers
            if task_type in p.supported_tasks and not p.is_last_resort
        ]
        candidates.sort(key=lambda p: (p.priority, -p.reliability_score, p.avg_latency_ms))

        failures = 0
        last_error = None

        for provider in candidates:
            if await self._is_in_cooldown(provider.id):
                continue

            api_key = await self.get_secret(provider.api_key_vault_ref)
            if not api_key:
                continue

            start = time.perf_counter()
            try:
                result = await self._call_openai_compatible(
                    provider=provider,
                    api_key=api_key,
                    messages=messages,
                    json_mode=json_mode,
                    temperature=temperature,
                )
                latency = (time.perf_counter() - start) * 1000
                self._update_score(provider, success=True, latency_ms=latency)
                return {
                    "content": result,
                    "provider_used": provider.id,
                    "model": provider.model,
                    "latency_ms": round(latency, 1),
                    "fallback_count": failures,
                }
            except httpx.HTTPStatusError as exc:
                failures += 1
                last_error = str(exc)
                self._update_score(provider, success=False, latency_ms=0)
                if exc.response.status_code in (429, 502, 503, 529):
                    await self._set_cooldown(provider.id, seconds=90)
            except Exception as exc:
                failures += 1
                last_error = str(exc)
                self._update_score(provider, success=False, latency_ms=0)
                await self._set_cooldown(provider.id, seconds=30)

        # Último recurso: Anthropic Claude API (somente se autorizado pelas travas)
        claude_node = next((p for p in self.providers if p.is_last_resort), None)
        if claude_node and await self._can_use_claude(failures, force_claude):
            claude_key = await self.get_secret(claude_node.api_key_vault_ref)
            if claude_key:
                start = time.perf_counter()
                result = await self._call_anthropic_native(
                    provider=claude_node,
                    api_key=claude_key,
                    messages=messages,
                    temperature=temperature,
                )
                await self._increment_claude_usage()
                latency = (time.perf_counter() - start) * 1000
                return {
                    "content": result,
                    "provider_used": claude_node.id,
                    "model": claude_node.model,
                    "latency_ms": round(latency, 1),
                    "fallback_count": failures,
                }

        raise RuntimeError(f"Todos os provedores falharam ou atingiram limite. Último erro: {last_error}")

    async def _call_openai_compatible(
        self,
        provider: ProviderNode,
        api_key: str,
        messages: List[Dict[str, Any]],
        json_mode: bool,
        temperature: float,
    ) -> str:
        headers = {
            "Authorization": f"Bearer {api_key}",
            "Content-Type": "application/json",
        }
        payload: Dict[str, Any] = {
            "model": provider.model,
            "messages": messages,
            "temperature": temperature,
        }
        if json_mode:
            payload["response_format"] = {"type": "json_object"}

        async with httpx.AsyncClient(timeout=35.0) as client:
            resp = await client.post(provider.base_url, headers=headers, json=payload)
            resp.raise_for_status()
            data = resp.json()
            return data["choices"][0]["message"]["content"]

    async def _call_anthropic_native(
        self,
        provider: ProviderNode,
        api_key: str,
        messages: List[Dict[str, Any]],
        temperature: float,
    ) -> str:
        system_prompt = ""
        user_messages = []
        for msg in messages:
            if msg["role"] == "system":
                system_prompt += msg["content"] + "\n"
            else:
                user_messages.append({"role": msg["role"], "content": msg["content"]})

        headers = {
            "x-api-key": api_key,
            "anthropic-version": "2023-06-01",
            "Content-Type": "application/json",
        }
        payload = {
            "model": provider.model,
            "max_tokens": 4096,
            "temperature": temperature,
            "system": system_prompt.strip(),
            "messages": user_messages,
        }
        async with httpx.AsyncClient(timeout=45.0) as client:
            resp = await client.post(provider.base_url, headers=headers, json=payload)
            resp.raise_for_status()
            data = resp.json()
            return data["content"][0]["text"]
